cmake_minimum_required(VERSION 3.16)
project(uran_ros2_test_plugin VERSION 1.0.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(pluginlib REQUIRED)
find_package(uran_plugin_api 2.1 REQUIRED CONFIG)
find_package(uran_ros2_adapter REQUIRED)
add_library(uran_ros2_test_plugins SHARED src/test_plugins.cpp)
target_compile_features(uran_ros2_test_plugins PUBLIC cxx_std_17)
target_include_directories(uran_ros2_test_plugins PRIVATE include)
target_link_libraries(uran_ros2_test_plugins
  uran::uran_plugin_api nlohmann_json::nlohmann_json)
ament_target_dependencies(uran_ros2_test_plugins pluginlib)
pluginlib_export_plugin_description_file(uran_ros2_adapter test_plugins.xml)
set_target_properties(uran_ros2_test_plugins PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS uran_ros2_test_plugins LIBRARY DESTINATION lib)
ament_package()
